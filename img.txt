services:
  proxy:
    image: nginx:1.27
    container_name: proxy
    ports:
      - "80:80"
    volumes:
      - ./proxy/default.conf:/etc/nginx/conf.d/default.conf:ro
    depends_on:
      - site1
      - site2
    networks:
      - appnet

  site1:
    image: nginx:1.27
    volumes:
      - ./site1:/usr/share/nginx/html:ro
    networks:
      - appnet

  site2:
    image: nginx:1.27
    volumes:
      - ./site2:/usr/share/nginx/html:ro
    networks:
      - appnet

networks:
  appnet:




proxy/default.conf (route by path):



server {
    listen 80;

    # Landing page at the root
    location = / {
        default_type text/html;
        return 200 '<h1>My Sites</h1><a href="/site1/">Site 1</a> | <a href="/site2/">Site 2</a>';
    }

    # Redirect missing trailing slash
    location = /site1 { return 301 /site1/; }
    location = /site2 { return 301 /site2/; }

    # Proxy to the site containers
    location /site1/ {
        proxy_pass http://site1/;
    }

    location /site2/ {
        proxy_pass http://site2/;
    }
}


cd ~/myapp
